# Lista de cuadrados
cuadrados = [n**2 for n in range(5)]
print(cuadrados)

Rust

                
  • Sistemas
  • WebAssembly
  • CLI

                
Velocidad de C con seguridad de memoria garantizada. El lenguaje más admirado de Stack Overflow 10 años seguidos.

                
main.rs
fn main() {
    let v = vec![1, 2, 3];
    let s: i32 = v.iter().sum();
    println!("Suma: {}", s);
}
